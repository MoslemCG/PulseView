cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" PULSEVIEW_PACKAGE_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/package.json")
string(JSON PULSEVIEW_VERSION GET "${PULSEVIEW_PACKAGE_JSON}" version)
project(PulseView VERSION ${PULSEVIEW_VERSION} LANGUAGES CXX)
configure_file(native/Version.h.in "${CMAKE_BINARY_DIR}/generated/Version.h" @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PULSEVIEW_BUILD_APP "Build the Windows desktop application" ${WIN32})
include(CTest)
if(BUILD_TESTING)
  add_executable(telemetry_math_tests tests/telemetry_math_tests.cpp)
  target_include_directories(telemetry_math_tests PRIVATE native)
  add_test(NAME telemetry_math COMMAND telemetry_math_tests)
endif()
if(PULSEVIEW_BUILD_APP)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The app target requires Windows, MSVC, and x64. Linux/macOS can run portable tests with -DPULSEVIEW_BUILD_APP=OFF.")
  endif()
  enable_language(RC)
  include(cmake/Dependencies.cmake)
  add_executable(PulseView WIN32
    native/main.cpp native/Application.cpp native/WebViewHost.cpp
    native/Telemetry.cpp native/PdhQuery.cpp native/Nvml.cpp
    native/Storage.cpp native/WinUtil.cpp native/app.rc)
  target_include_directories(PulseView PRIVATE native "${CMAKE_BINARY_DIR}/generated" "${PULSEVIEW_JSON_INCLUDE_DIR}" "${PULSEVIEW_WEBVIEW2_INCLUDE}")
  target_compile_definitions(PulseView PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(PulseView PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(PulseView PRIVATE /MANIFEST:NO)
  target_link_libraries(PulseView PRIVATE "${PULSEVIEW_WEBVIEW2_LOADER}"
    user32 gdi32 shell32 shlwapi ole32 oleaut32 uuid advapi32 dwmapi dcomp dxgi
    pdh psapi iphlpapi ws2_32 version)
  set_target_properties(PulseView PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  # Always copy on build, even when only HTML/CSS/JS changed.
  add_custom_target(pulseview_ui ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:PulseView>/ui"
    COMMENT "Copying local UI assets (no Node/bundler dependency)")
  add_dependencies(PulseView pulseview_ui)
  if(BUILD_TESTING)
    add_executable(storage_validation_tests tests/storage_validation_tests.cpp native/WinUtil.cpp)
    target_include_directories(storage_validation_tests PRIVATE native "${PULSEVIEW_JSON_INCLUDE_DIR}")
    target_compile_definitions(storage_validation_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(storage_validation_tests PRIVATE /utf-8 /EHsc)
    target_link_libraries(storage_validation_tests PRIVATE shell32 ole32)
    add_test(NAME storage_validation COMMAND storage_validation_tests)
  endif()
  install(TARGETS PulseView RUNTIME DESTINATION .)
  # App-local MSVC runtime: users do not need a compiler or a separate VC install.
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
  install(DIRECTORY ui DESTINATION .)
  install(FILES README.md README.fa.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY examples DESTINATION .)
  file(GLOB WV_LICENSES "${PULSEVIEW_WEBVIEW2_DIR}/*LICENSE*" "${PULSEVIEW_WEBVIEW2_DIR}/*License*")
  if(WV_LICENSES)
    install(FILES ${WV_LICENSES} DESTINATION licenses/WebView2)
  endif()
  if(EXISTS "${CMAKE_BINARY_DIR}/_deps/json-3.12.0/LICENSE.MIT")
    install(FILES "${CMAKE_BINARY_DIR}/_deps/json-3.12.0/LICENSE.MIT" DESTINATION licenses/nlohmann-json)
  elseif(EXISTS "${PULSEVIEW_JSON_INCLUDE_DIR}/../LICENSE.MIT")
    install(FILES "${PULSEVIEW_JSON_INCLUDE_DIR}/../LICENSE.MIT" DESTINATION licenses/nlohmann-json)
  endif()
endif()
